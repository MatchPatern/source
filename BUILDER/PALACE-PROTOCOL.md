To achieve Level 4 on a file, you must document its Beginning, Middle, and End structure:

Section What to Identify

M: Middle Core argument, the "how-to," the central transmission
E: End Final section, call to action, concluding integration

Surgical Strike Protocol: For large master files (e.g., pillars.md), you only need to map the specific section relevant to the current mission. Document the B-M-E of that section.

3.3 Auto-Upgrade Logic

```python
def upgrade_fidelity(visit_count: int, current_level: int) -> int:
    if visit_count >= 10 and current_level < 1: return 1
    elif visit_count >= 25 and current_level < 2: return 2
    elif visit_count >= 50 and current_level < 3: return 3
    elif visit_count >= 100 and current_level < 4: return 4
    elif visit_count >= 200 and current_level < 5: return 5
    return current_level

# observer.py — watches the palace and auto-updates fidelity
import os
import json
from pathlib import Path
from datetime import datetime

class PalaceObserver:
    def __init__(self, palace_path):
        self.palace_path = Path(palace_path)
        self.state_file = self.palace_path / "BUILDER" / "OBSERVER-STATE.json"
        self.load_state()
    
    def load_state(self):
        if self.state_file.exists():
            with open(self.state_file) as f:
                self.state = json.load(f)
        else:
            self.state = {"files": {}}
    
    def scan(self):
        """Scan all markdown files for headers"""
        for file in self.palace_path.rglob("*.md"):
            header = self.extract_header(file)
            file_key = str(file.relative_to(self.palace_path))
            if file_key not in self.state["files"]:
                self.state["files"][file_key] = {
                    "visits": 0,
                    "fidelity": "L0",
                    "last_visit": None
                }
            # Check if file was modified since last scan
            mtime = os.path.getmtime(file)
            if mtime > self.state["files"][file_key].get("last_modified", 0):
                # File was changed — increment visits
                self.state["files"][file_key]["visits"] += 1
                self.state["files"][file_key]["last_visit"] = datetime.now().isoformat()
                self.state["files"][file_key]["last_modified"] = mtime
                self.upgrade_fidelity(file_key)
    
    def upgrade_fidelity(self, file_key):
        visits = self.state["files"][file_key]["visits"]
        if visits >= 200:
            new_level = "L5"
        elif visits >= 100:
            new_level = "L4"
        elif visits >= 50:
            new_level = "L3"
        elif visits >= 25:
            new_level = "L2"
        elif visits >= 10:
            new_level = "L1"
        else:
            new_level = "L0"
        self.state["files"][file_key]["fidelity"] = new_level
    
    def save_state(self):
        with open(self.state_file, "w") as f:
            json.dump(self.state, f, indent=2)
```
